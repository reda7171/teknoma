+/**
+ * Mettre ├á jour les articles d'une commande (Edition dynamique)
+ */
+export async function updateOrderItems(orderId: string, items: {
+    id?: string;
+    type: 'BOOK' | 'PACK';
+    productId: string;
+    quantity: number;
+    price: number;
+}[]) {
+    try {
+        await verifyAdmin()
+
+        const order = await prisma.order.findUnique({
+            where: { id: orderId },
+            include: { items: true }
+        })
+
+        if (!order) {
+            throw new Error('Commande introuvable')
+        }
+
+        // Si la commande n'a pas un statut qui a d├®cr├®ment├® le stock, on n'impacte pas le stock
+        // Habituellement, toute commande non annul├®e a d├®cr├®ment├® le stock ├á la cr├®ation.
+        // Si elle est "RETURNED" ou "CANCELED" ou "CANCELLED", le stock a d├®j├á ├®t├® r├®int├®gr├®.
+        const isCanceledOrReturned = order.status === 'CANCELED' || order.status === 'CANCELLED' || order.status === 'RETURNED'
+        const impactsStock = !isCanceledOrReturned
+
+        let newSubtotal = 0
+
+        // Transaction pour assurer la coh├®rence
+        await prisma.$transaction(async (tx) => {
+            // 1. Pour chaque ancien item, on regarde s'il est supprim├® ou modifi├®
+            for (const oldItem of order.items) {
+                const newItem = items.find(i => i.id === oldItem.id)
+                
+                if (!newItem) {
+                    // Item supprim├®
+                    if (impactsStock) {
+                        if (oldItem.type === 'BOOK' && oldItem.bookId) {
+                            await tx.book.update({
+                                where: { id: oldItem.bookId },
+                                data: { stock: { increment: oldItem.quantity } }
+                            })
+                        } else if (oldItem.type === 'PACK' && oldItem.packId) {
+                            const packBooks = await tx.packBook.findMany({ where: { packId: oldItem.packId } })
+                            for (const pb of packBooks) {
+                                await tx.book.update({
+                                    where: { id: pb.bookId },
+                                    data: { stock: { increment: oldItem.quantity } }
+                                })
+                            }
+                        }
+                    }
+                    await tx.orderItem.delete({ where: { id: oldItem.id } })
+                } else {
+                    // Item modifi├® (changement de quantit├®)
+                    const diff = newItem.quantity - oldItem.quantity
+                    if (diff !== 0 && impactsStock) {
+                        if (oldItem.type === 'BOOK' && oldItem.bookId) {
+                            await tx.book.update({
+                                where: { id: oldItem.bookId },
+                                data: { stock: { decrement: diff } }
+                            })
+                        } else if (oldItem.type === 'PACK' && oldItem.packId) {
+                            const packBooks = await tx.packBook.findMany({ where: { packId: oldItem.packId } })
+                            for (const pb of packBooks) {
+                                await tx.book.update({
+                                    where: { id: pb.bookId },
+                                    data: { stock: { decrement: diff } }
+                                })
+                            }
+                        }
+                    }
+                    
+                    // Calculer le prix
+                    newSubtotal += newItem.price * newItem.quantity
+                    
+                    await tx.orderItem.update({
+                        where: { id: oldItem.id },
+                        data: {
+                            quantity: newItem.quantity,
+                            price: newItem.price,
+                        }
+                    })
+                }
+            }
+
+            // 2. Pour chaque nouvel item, on l'ajoute
+            for (const newItem of items.filter(i => !i.id)) {
+                let costPrice = 0
+                if (newItem.type === 'BOOK') {
+                    const book = await tx.book.findUnique({ where: { id: newItem.productId }})
+                    if (book) {
+                        costPrice = book.costPrice
+                        if (impactsStock) {
+                            await tx.book.update({
+                                where: { id: newItem.productId },
+                                data: { stock: { decrement: newItem.quantity } }
+                            })
+                        }
+                    }
+                } else if (newItem.type === 'PACK') {
+                    const pack = await tx.pack.findUnique({ where: { id: newItem.productId }})
+                    if (pack) {
+                        costPrice = pack.costPrice
+                        if (impactsStock) {
+                            const packBooks = await tx.packBook.findMany({ where: { packId: newItem.productId } })
+                            for (const pb of packBooks) {
+                                await tx.book.update({
+                                    where: { id: pb.bookId },
+                                    data: { stock: { decrement: newItem.quantity } }
+                                })
+                            }
+                        }
+                    }
+                }
+
+                newSubtotal += newItem.price * newItem.quantity
+
+                await tx.orderItem.create({
+                    data: {
+                        orderId: order.id,
+                        type: newItem.type as any,
+                        quantity: newItem.quantity,
+                        price: newItem.price,
+                        costPrice: costPrice,
+                        bookId: newItem.type === 'BOOK' ? newItem.productId : null,
+                        packId: newItem.type === 'PACK' ? newItem.productId : null
+                    }
+                })
+            }
+
+            // 3. Mettre ├á jour la commande
+            const newTotal = newSubtotal + order.shippingFees - order.discount
+            await tx.order.update({
+                where: { id: order.id },
+                data: {
+                    subtotal: newSubtotal,
+                    total: newTotal,
+                    updatedAt: new Date()
+                }
+            })
+
+            // 4. Historique
+            await tx.orderStatusHistory.create({
+                data: {
+                    orderId: order.id,
+                    status: order.status,
+                    comment: "Modification des articles et quantit├®s de la commande",
+                    createdBy: "ADMIN"
+                }
+            })
+            
+            // Log global
+            await tx.auditLog.create({
+                data: {
+                    action: 'UPDATE_ITEMS',
+                    entity: 'ORDER',
+                    entityId: order.id,
+                    details: `Mise ├á jour des articles. Nouveau total: ${newTotal}`,
+                    userId: 'SYSTEM'
+                }
+            })
+        })
+
+        revalidatePath('/admin/orders')
+        revalidatePath(`/admin/orders/${orderId}`)
+
+        return {
+            success: true,
+            message: "Les articles de la commande ont ├®t├® mis ├á jour"
+        }
+    } catch (error: any) {
+        console.error('Erreur lors de la modification des articles:', error)
+        return {
+            success: false,
+            error: error.message || 'Impossible de mettre ├á jour les articles'
+        }
+    }
